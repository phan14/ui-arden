    var desc = document.getElementById('arden-detail-desc');

    var equip1 = document.getElementById('arden-equip-1');
    var equip2 = document.getElementById('arden-equip-2');
    var equip3 = document.getElementById('arden-equip-3');


    function changeDetail(index){

        var item = data[index];

        if(!item) return;

        cards.forEach(function(card){
            card.classList.remove('is-active');
        });

        cards[index].classList.add('is-active');

        title.textContent = item.title;
        capacity.textContent = 'Công suất: ' + item.capacity;
        desc.textContent = item.desc;

        equip1.textContent = item.equip[0];
        equip2.textContent = item.equip[1];
        equip3.textContent = item.equip[2];

    }


    cards.forEach(function(card, index){

        card.setAttribute('tabindex','0');
        card.setAttribute('role','button');

        card.addEventListener('click', function(){
            changeDetail(index);
        });

        card.addEventListener('keydown', function(event){

            if(event.key === 'Enter' || event.key === ' '){
                event.preventDefault();
                changeDetail(index);
            }

        });

    });

});
